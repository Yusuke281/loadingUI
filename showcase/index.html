Loading UI Showcase

A collection of loading indicators from the experiment project.

Spinner

/* HTML */
<div class="spinner"></div>

/* CSS */
.spinner {
    width: 48px;
    height: 48px;
    border: 4px solid #f3f3f3;
    border-top: 4px solid #007bff;
    border-radius: 50%;
    animation: spin 1s linear infinite;
}

@keyframes spin {
    0% { transform: rotate(0deg); }
    100% { transform: rotate(360deg); }
}

Skeleton

/* HTML */
<div class="skeleton-card">
    <div class="skeleton-image"></div>
    <div class="skeleton-content">
        <div class="skeleton-text"></div>
        <div class="skeleton-text short"></div>
    </div>
    <div class="skeleton-button"></div>
</div>

/* CSS */
.skeleton-card {
    background: #2a2a2a;
    border-radius: 8px;
    overflow: hidden;
    display: flex;
    flex-direction: column;
    width: 100%;
    height: 100%;
}
.skeleton-image, .skeleton-text, .skeleton-button {
    background: linear-gradient(90deg, #3c3c3c 25%, #444 50%, #3c3c3c 75%);
    background-size: 200% 100%;
    animation: shimmer 1.5s infinite;
}
.skeleton-image { height: 50%; }
.skeleton-content { padding: 1rem; flex-grow: 1; }
.skeleton-text { height: 1rem; border-radius: 4px; margin-bottom: 0.5rem; }
.skeleton-text.short { width: 60%; }
.skeleton-button { height: 30px; margin: 1rem; margin-top: auto; border-radius: 5px; }

@keyframes shimmer {
    0% { background-position: 200% 0; }
    100% { background-position: -200% 0; }
}
